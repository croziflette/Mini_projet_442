static uint16_t ReadLE16(const uint8_t *p)
{
  return (uint16_t)(p[0] | (p[1] << 8));
}

static int32_t ReadLE32(const uint8_t *p)
{
  return (int32_t)(p[0] | (p[1] << 8) | (p[2] << 16) | (p[3] << 24));
}

static uint8_t RGB565_ToGray(uint16_t c)
{
  uint32_t r = (c >> 11) & 0x1FU;
  uint32_t g = (c >> 5)  & 0x3FU;
  uint32_t b =  c        & 0x1FU;

  r = (r * 255U) / 31U;
  g = (g * 255U) / 63U;
  b = (b * 255U) / 31U;

  return (uint8_t)((77U * r + 150U * g + 29U * b) >> 8);
}

static void BMP565_ToGrayBuffer(const uint8_t *bmp,
                                uint8_t *dst,
                                uint32_t dstWidth,
                                uint32_t dstHeight)
{
  uint32_t dataOffset = (uint32_t)ReadLE32(&bmp[10]);
  int32_t bmpWidthSigned = ReadLE32(&bmp[18]);
  int32_t bmpHeightSigned = ReadLE32(&bmp[22]);
  uint16_t bpp = ReadLE16(&bmp[28]);

  uint32_t bmpWidth;
  uint32_t bmpHeight;
  uint32_t rowSize;
  uint32_t x, y;
  uint8_t isBottomUp;

  for (y = 0; y < dstHeight; y++)
  {
    for (x = 0; x < dstWidth; x++)
    {
      dst[y * dstWidth + x] = 0U;
    }
  }

  if (bmpWidthSigned <= 0 || bmpHeightSigned == 0 || bpp != 16U)
    return;

  bmpWidth = (uint32_t)bmpWidthSigned;
  bmpHeight = (bmpHeightSigned > 0) ? (uint32_t)bmpHeightSigned : (uint32_t)(-bmpHeightSigned);
  isBottomUp = (bmpHeightSigned > 0) ? 1U : 0U;

  bmpWidth = MIN2(bmpWidth, dstWidth);
  bmpHeight = MIN2(bmpHeight, dstHeight);

  rowSize = ((bmpWidth * 16U + 31U) / 32U) * 4U;

  for (y = 0; y < bmpHeight; y++)
  {
    uint32_t srcRow = isBottomUp ? (bmpHeight - 1U - y) : y;
    const uint8_t *line = bmp + dataOffset + srcRow * rowSize;

    for (x = 0; x < bmpWidth; x++)
    {
      uint16_t pix = ReadLE16(&line[2U * x]);
      dst[y * dstWidth + x] = RGB565_ToGray(pix);
    }
  }
}

static void GaussianBlur3x3(const uint8_t *src,
                            uint8_t *dst,
                            uint32_t width,
                            uint32_t height)
{
  uint32_t x, y;

  for (y = 0; y < height; y++)
  {
    for (x = 0; x < width; x++)
    {
      uint32_t idx = y * width + x;

      if (x == 0U || y == 0U || x == (width - 1U) || y == (height - 1U))
      {
        dst[idx] = src[idx];
      }
      else
      {
        uint32_t sum =
            src[idx - width - 1U] + 2U * src[idx - width] + src[idx - width + 1U] +
            2U * src[idx - 1U]     + 4U * src[idx]         + 2U * src[idx + 1U] +
            src[idx + width - 1U]  + 2U * src[idx + width] + src[idx + width + 1U];

        dst[idx] = (uint8_t)(sum >> 4);
      }
    }
  }
}

static void ComputeLaplacian4(const uint8_t *src,
                              int16_t *lap,
                              uint32_t width,
                              uint32_t height)
{
  uint32_t x, y;

  for (y = 0; y < height; y++)
  {
    for (x = 0; x < width; x++)
    {
      uint32_t idx = y * width + x;

      if (x == 0U || y == 0U || x == (width - 1U) || y == (height - 1U))
      {
        lap[idx] = 0;
      }
      else
      {
        lap[idx] =
            (int16_t)src[idx - 1U] +
            (int16_t)src[idx + 1U] +
            (int16_t)src[idx - width] +
            (int16_t)src[idx + width] -
            4 * (int16_t)src[idx];
      }
    }
  }
}

static void DetectZeroCrossings2x2(const int16_t *lap,
                                   uint8_t *zmask,
                                   uint32_t width,
                                   uint32_t height,
                                   int16_t contrastTh)
{
  uint32_t x, y;

  for (y = 0; y < height; y++)
  {
    for (x = 0; x < width; x++)
    {
      zmask[y * width + x] = 0U;
    }
  }

  for (y = 0; y < height - 1U; y++)
  {
    for (x = 0; x < width - 1U; x++)
    {
      uint32_t idx = y * width + x;

      int16_t a = lap[idx];
      int16_t b = lap[idx + 1U];
      int16_t c = lap[idx + width];
      int16_t d = lap[idx + width + 1U];

      int16_t minv = a;
      int16_t maxv = a;

      if (b < minv) minv = b;
      if (c < minv) minv = c;
      if (d < minv) minv = d;

      if (b > maxv) maxv = b;
      if (c > maxv) maxv = c;
      if (d > maxv) maxv = d;

      if ((minv < 0) && (maxv > 0) && ((maxv - minv) >= contrastTh))
      {
        zmask[idx] = 1U;
        zmask[idx + 1U] = 1U;
        zmask[idx + width] = 1U;
        zmask[idx + width + 1U] = 1U;
      }
    }
  }
}

static void BuildGradientHysteresisMask(const uint8_t *src,
                                        uint8_t *weakMask,
                                        uint8_t *edgeMask,
                                        uint32_t *queue,
                                        uint32_t width,
                                        uint32_t height,
                                        uint16_t lowTh,
                                        uint16_t highTh)
{
  uint32_t x, y;
  uint32_t head = 0U;
  uint32_t tail = 0U;

  for (y = 0; y < height; y++)
  {
    for (x = 0; x < width; x++)
    {
      uint32_t idx = y * width + x;
      weakMask[idx] = 0U;
      edgeMask[idx] = 0U;
    }
  }

  for (y = 1U; y < height - 1U; y++)
  {
    for (x = 1U; x < width - 1U; x++)
    {
      uint32_t idx = y * width + x;

      int gx =
          -(int)src[idx - width - 1U] + (int)src[idx - width + 1U] +
          -2 * (int)src[idx - 1U]     + 2 * (int)src[idx + 1U] +
          -(int)src[idx + width - 1U] + (int)src[idx + width + 1U];

      int gy =
          -(int)src[idx - width - 1U] - 2 * (int)src[idx - width] - (int)src[idx - width + 1U] +
           (int)src[idx + width - 1U] + 2 * (int)src[idx + width] + (int)src[idx + width + 1U];

      uint16_t mag = (uint16_t)(ABS(gx) + ABS(gy));

      if (mag >= lowTh)
      {
        weakMask[idx] = 1U;

        if (mag >= highTh)
        {
          edgeMask[idx] = 1U;
          queue[tail++] = idx;
        }
      }
    }
  }

  while (head < tail)
  {
    uint32_t idx = queue[head++];
    uint32_t cx = idx % width;
    uint32_t cy = idx / width;
    int32_t dx, dy;

    for (dy = -1; dy <= 1; dy++)
    {
      for (dx = -1; dx <= 1; dx++)
      {
        if (dx == 0 && dy == 0)
          continue;

        int32_t nx = (int32_t)cx + dx;
        int32_t ny = (int32_t)cy + dy;

        if (nx >= 0 && nx < (int32_t)width && ny >= 0 && ny < (int32_t)height)
        {
          uint32_t nidx = (uint32_t)ny * width + (uint32_t)nx;

          if (weakMask[nidx] && !edgeMask[nidx])
          {
            edgeMask[nidx] = 1U;
            queue[tail++] = nidx;
          }
        }
      }
    }
  }
}

static void RenderFinalContours(const uint8_t *zmask,
                                const uint8_t *edgeMask,
                                uint32_t *dstArgb,
                                uint32_t width,
                                uint32_t height)
{
  uint32_t x, y;

  for (y = 0; y < height; y++)
  {
    for (x = 0; x < width; x++)
    {
      uint32_t idx = y * width + x;

      if (zmask[idx] && edgeMask[idx])
        dstArgb[idx] = 0xFFFF0000U;
      else
        dstArgb[idx] = 0x00000000U;
    }
  }
}

/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{
  /* USER CODE BEGIN 1 */
  char text[80] = {0};
  uint32_t potr;
  uint8_t *grayBuf = (uint8_t *)GRAY_ADDRESS;
  uint8_t *smoothBuf = (uint8_t *)SMOOTH_ADDRESS;
  uint8_t *zeroMask = (uint8_t *)ZERO_ADDRESS;
  uint8_t *weakMask = (uint8_t *)WEAK_ADDRESS;
  uint8_t *edgeMask = (uint8_t *)EDGE_ADDRESS;
  int16_t *lapBuf = (int16_t *)LAPLACIAN_ADDRESS;
  uint32_t *queueBuf = (uint32_t *)QUEUE_ADDRESS;
  uint32_t *layer1Buf = (uint32_t *)LAYER1_ADDRESS;

  ADC_ChannelConfTypeDef sConfig = {0};
  sConfig.Channel = ADC_CHANNEL_6;
  sConfig.Rank = ADC_REGULAR_RANK_1;
  sConfig.SamplingTime = ADC_SAMPLETIME_3CYCLES;
  /* USER CODE END 1 */

  /* MCU Configuration--------------------------------------------------------*/
  HAL_Init();
  SystemClock_Config();

  /* Initialize all configured peripherals */
  MX_GPIO_Init();
  MX_ADC3_Init();
  MX_DMA2D_Init();
  MX_FMC_Init();
  MX_I2C1_Init();
  MX_I2C3_Init();
  MX_LTDC_Init();
  MX_RTC_Init();
  MX_SPI2_Init();
  MX_TIM1_Init();
  MX_TIM2_Init();
  MX_TIM3_Init();
  MX_TIM5_Init();
  MX_TIM8_Init();
  MX_USART1_UART_Init();
  MX_USART6_UART_Init();
  MX_ADC1_Init();
  MX_DAC_Init();
  MX_UART7_Init();

  /* USER CODE BEGIN 2 */
  BSP_LCD_Init();

  BSP_LCD_LayerDefaultInit(0, LAYER0_ADDRESS);
  BSP_LCD_LayerDefaultInit(1, LAYER1_ADDRESS);
  BSP_LCD_DisplayOn();

  BSP_LCD_SelectLayer(0);
  BSP_LCD_Clear(LCD_COLOR_BLACK);
  BSP_LCD_DrawBitmap(0, 0, (uint8_t *)SELECTED_BMP);

  BSP_LCD_SelectLayer(1);
  BSP_LCD_Clear(0x00000000U);
  BSP_LCD_SetFont(&Font12);
  BSP_LCD_SetTextColor(LCD_COLOR_BLUE);
  BSP_LCD_SetBackColor(0x00000000U);

  HAL_ADC_ConfigChannel(&hadc3, &sConfig);

  /* Pré-calculs indépendants du potentiomètre */
  BMP565_ToGrayBuffer((const uint8_t *)SELECTED_BMP, grayBuf, LCD_WIDTH, LCD_HEIGHT);
  GaussianBlur3x3(grayBuf, smoothBuf, LCD_WIDTH, LCD_HEIGHT);
  ComputeLaplacian4(smoothBuf, lapBuf, LCD_WIDTH, LCD_HEIGHT);
  DetectZeroCrossings2x2(lapBuf, zeroMask, LCD_WIDTH, LCD_HEIGHT, LAPLACIAN_CONTRAST_TH);

  /* Première lecture du potentiomètre */
  HAL_ADC_Start(&hadc3);
  while (HAL_ADC_PollForConversion(&hadc3, 100) != HAL_OK)
  {
  }
  potr = HAL_ADC_GetValue(&hadc3);
  HAL_ADC_Stop(&hadc3);

  high_threshold = HYST_HIGH_MIN + (potr * (HYST_HIGH_MAX - HYST_HIGH_MIN)) / 4095U;
  low_threshold = high_threshold / 3U;
  if (low_threshold < 1U)
    low_threshold = 1U;

  BuildGradientHysteresisMask(smoothBuf,
                              weakMask,
                              edgeMask,
                              queueBuf,
                              LCD_WIDTH,
                              LCD_HEIGHT,
                              (uint16_t)low_threshold,
                              (uint16_t)high_threshold);

  RenderFinalContours(zeroMask, edgeMask, layer1Buf, LCD_WIDTH, LCD_HEIGHT);

  BSP_LCD_SelectLayer(1);
  sprintf(text, "H:%lu  L:%lu  LapC:%d   ", (unsigned long)high_threshold,
                                          (unsigned long)low_threshold,
                                          LAPLACIAN_CONTRAST_TH);
  BSP_LCD_DisplayStringAtLine(1, (uint8_t *)text);

  last_high_threshold = high_threshold;
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    HAL_ADC_Start(&hadc3);
    while (HAL_ADC_PollForConversion(&hadc3, 100) != HAL_OK)
    {
    }
    potr = HAL_ADC_GetValue(&hadc3);
    HAL_ADC_Stop(&hadc3);

    high_threshold = HYST_HIGH_MIN + (potr * (HYST_HIGH_MAX - HYST_HIGH_MIN)) / 4095U;
    low_threshold = high_threshold / 3U;
    if (low_threshold < 1U)
      low_threshold = 1U;

    if (high_threshold != last_high_threshold)
    {
      BuildGradientHysteresisMask(smoothBuf,
                                  weakMask,
                                  edgeMask,
                                  queueBuf,
                                  LCD_WIDTH,
                                  LCD_HEIGHT,
                                  (uint16_t)low_threshold,
                                  (uint16_t)high_threshold);

      RenderFinalContours(zeroMask, edgeMask, layer1Buf, LCD_WIDTH, LCD_HEIGHT);

      BSP_LCD_SelectLayer(1);
      BSP_LCD_SetFont(&Font12);
      BSP_LCD_SetTextColor(LCD_COLOR_BLUE);
      BSP_LCD_SetBackColor(0x00000000U);

      sprintf(text, "H:%lu  L:%lu  LapC:%d   ", (unsigned long)high_threshold,
                                              (unsigned long)low_threshold,
                                              LAPLACIAN_CONTRAST_TH);
      BSP_LCD_DisplayStringAtLine(1, (uint8_t *)text);

      last_high_threshold = high_threshold;
    }

    HAL_Delay(20);
  }
  /* USER CODE END WHILE */

  /* USER CODE BEGIN 3 */
  /* USER CODE END 3 */
}

/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  HAL_PWR_EnableBkUpAccess();
  __HAL_RCC_PWR_CLK_ENABLE();
  __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_LSI | RCC_OSCILLATORTYPE_HSE;
  RCC_OscInitStruct.HSEState = RCC_HSE_ON;
  RCC_OscInitStruct.LSIState = RCC_LSI_ON;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSE;
  RCC_OscInitStruct.PLL.PLLM = 25;
  RCC_OscInitStruct.PLL.PLLN = 400;
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV2;
  RCC_OscInitStruct.PLL.PLLQ = 9;

  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  if (HAL_PWREx_EnableOverDrive() != HAL_OK)
  {
    Error_Handler();
  }

  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                              | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV4;
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV2;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_6) != HAL_OK)
  {
    Error_Handler();
  }
}

/**
  * @brief  Period elapsed callback in non blocking mode
  * @note   This function is called when TIM6 interrupt took place, inside
  * HAL_TIM_IRQHandler(). It makes a direct call to HAL_IncTick() to increment
  * a global variable "uwTick" used as application time base.
  * @param  htim : TIM handle
  * @retval None
  */
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  if (htim->Instance == TIM6)
  {
    HAL_IncTick();
  }
}

/**
  * @brief  This function is executed in case of error occurrence.
  * @retval None
  */

{
  __disable_irq();
  while (1)
  {
  }
}
