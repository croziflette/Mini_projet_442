static uint8_t RGB565_ToGray(uint16_t c)
{
  uint32_t r = (c >> 11) & 0x1FU;
  uint32_t g = (c >> 5)  & 0x3FU;
  uint32_t b =  c        & 0x1FU;

  r = (r * 255U) / 31U;
  g = (g * 255U) / 63U;
  b = (b * 255U) / 31U;

  return (uint8_t)((77U * r + 150U * g + 29U * b) >> 8);
}

static void RGB565_ToRGB888(uint16_t c, uint8_t *r, uint8_t *g, uint8_t *b)
{
  uint32_t rr = (c >> 11) & 0x1FU;
  uint32_t gg = (c >> 5)  & 0x3FU;
  uint32_t bb =  c        & 0x1FU;

  *r = (uint8_t)((rr * 255U) / 31U);
  *g = (uint8_t)((gg * 255U) / 63U);
  *b = (uint8_t)((bb * 255U) / 31U);
}

static uint8_t IsSkinPixelRGB565(uint16_t c)
{
  uint8_t r, g, b;
  int32_t cb, cr;
  uint32_t y;

  RGB565_ToRGB888(c, &r, &g, &b);

  y  = (77U * r + 150U * g + 29U * b) >> 8;
  if (y < 35U)
  {
    return 0U;
  }

  /* Approximation YCbCr entière */
  cb = 128 + (((-43 * (int32_t)r) - (85 * (int32_t)g) + (128 * (int32_t)b)) >> 8);
  cr = 128 + (((128 * (int32_t)r) - (107 * (int32_t)g) - (21 * (int32_t)b)) >> 8);

  /* Heuristique peau */
  if ((cb >= 77) && (cb <= 127) &&
      (cr >= 133) && (cr <= 173))
  {
    return 1U;
  }

  return 0U;
}

static void CameraRGB565_GetGrayRow(const uint16_t *frame,
                                    uint32_t row,
                                    uint32_t width,
                                    uint8_t *dst)
{
  uint32_t x;
  const uint16_t *line = frame + row * width;

  for (x = 0; x < width; x++)
  {
    dst[x] = RGB565_ToGray(line[x]);
  }
}

static void ApplySobelToCameraRGB565(const uint16_t *frame,
                                     uint32_t *dst,
                                     uint32_t width,
                                     uint32_t height,
                                     uint32_t threshold,
                                     uint32_t color)
{
  uint8_t *l0 = gray_line_0;
  uint8_t *l1 = gray_line_1;
  uint8_t *l2 = gray_line_2;
  uint8_t *tmp;
  uint32_t x, y;

  if ((width < 3U) || (height < 3U))
  {
    return;
  }

  for (y = 0; y < height; y++)
  {
    for (x = 0; x < width; x++)
    {
      dst[y * width + x] = 0x00000000U;
    }
  }

  CameraRGB565_GetGrayRow(frame, 0U, width, l0);
  CameraRGB565_GetGrayRow(frame, 1U, width, l1);
  CameraRGB565_GetGrayRow(frame, 2U, width, l2);

  for (x = 0; x < width; x++)
  {
    dst[x] = 0x00000000U;
    dst[(height - 1U) * width + x] = 0x00000000U;
  }

  for (y = 0; y < height; y++)
  {
    dst[y * width] = 0x00000000U;
    dst[y * width + (width - 1U)] = 0x00000000U;
  }

  for (y = 1U; y < height - 1U; y++)
  {
    for (x = 1U; x < width - 1U; x++)
    {
      int gx =
          -l0[x - 1U] + l0[x + 1U]
          -2 * l1[x - 1U] + 2 * l1[x + 1U]
          -l2[x - 1U] + l2[x + 1U];

      int gy =
          -l0[x - 1U] - 2 * l0[x] - l0[x + 1U]
          +l2[x - 1U] + 2 * l2[x] + l2[x + 1U];

      uint32_t mag = (uint32_t)(ABS(gx) + ABS(gy));

      if (mag > threshold)
      {
        dst[y * width + x] = color;
      }
      else
      {
        dst[y * width + x] = 0x00000000U;
      }
    }

    if (y < height - 2U)
    {
      tmp = l0;
      l0 = l1;
      l1 = l2;
      l2 = tmp;

      CameraRGB565_GetGrayRow(frame, y + 2U, width, l2);
    }
  }
}

static void LCD_LL_ConvertLineToARGB8888(void *pSrc, void *pDst)
{
  hdma2d.Instance = DMA2D;

  hdma2d.Init.Mode         = DMA2D_M2M_PFC;
  hdma2d.Init.ColorMode    = DMA2D_ARGB8888;
  hdma2d.Init.OutputOffset = 0;

  hdma2d.LayerCfg[1].AlphaMode      = DMA2D_NO_MODIF_ALPHA;
  hdma2d.LayerCfg[1].InputAlpha     = 0xFF;
  hdma2d.LayerCfg[1].InputColorMode = CM_RGB565;
  hdma2d.LayerCfg[1].InputOffset    = 0;

  if (HAL_DMA2D_Init(&hdma2d) != HAL_OK)
  {
    Error_Handler();
  }

  if (HAL_DMA2D_ConfigLayer(&hdma2d, 1) != HAL_OK)
  {
    Error_Handler();
  }

  if (HAL_DMA2D_Start(&hdma2d,
                      (uint32_t)pSrc,
                      (uint32_t)pDst,
                      LCD_WIDTH,
                      1) != HAL_OK)
  {
    Error_Handler();
  }

  if (HAL_DMA2D_PollForTransfer(&hdma2d, 100) != HAL_OK)
  {
    Error_Handler();
  }
}

static void CameraFrameToLayer0(void)
{
  uint32_t y;
  uint32_t src = CAMERA_FRAME_BUFFER;
  uint32_t dst = LAYER0_ADDRESS;

  for (y = 0U; y < LCD_HEIGHT; y++)
  {
    LCD_LL_ConvertLineToARGB8888((void *)src, (void *)dst);
    src += LCD_WIDTH * 2U;
    dst += LCD_WIDTH * 4U;
  }
}

static void ReadSobelThresholdFromPot(void)
{
  uint32_t potr;

  HAL_ADC_Start(&hadc3);
  while (HAL_ADC_PollForConversion(&hadc3, 100) != HAL_OK)
  {
  }
  potr = HAL_ADC_GetValue(&hadc3);
  HAL_ADC_Stop(&hadc3);

  sobel_threshold = SOBEL_MIN_TH + (potr * (SOBEL_MAX_TH - SOBEL_MIN_TH)) / 4095U;
}

static void BuildSkinMaskFromFrozenImage(const uint16_t *frame)
{
  uint32_t x, y;

  for (y = 0U; y < FACE_H; y++)
  {
    for (x = 0U; x < FACE_W; x++)
    {
      uint32_t sx = x * FACE_DS + (FACE_DS / 2U);
      uint32_t sy = y * FACE_DS + (FACE_DS / 2U);

      if (sx >= LCD_WIDTH)
      {
        sx = LCD_WIDTH - 1U;
      }
      if (sy >= LCD_HEIGHT)
      {
        sy = LCD_HEIGHT - 1U;
      }

      skin_mask[y * FACE_W + x] = IsSkinPixelRGB565(frame[sy * LCD_WIDTH + sx]);
    }
  }
}

static void FilterSkinMask(void)
{
  uint32_t x, y;

  for (y = 0U; y < FACE_H; y++)
  {
    for (x = 0U; x < FACE_W; x++)
    {
      uint32_t idx = y * FACE_W + x;

      if ((x == 0U) || (y == 0U) || (x == (FACE_W - 1U)) || (y == (FACE_H - 1U)))
      {
        skin_mask_filt[idx] = 0U;
      }
      else
      {
        uint32_t count = 0U;
        int32_t dx, dy;

        for (dy = -1; dy <= 1; dy++)
        {
          for (dx = -1; dx <= 1; dx++)
          {
            uint32_t nidx = (uint32_t)((int32_t)y + dy) * FACE_W + (uint32_t)((int32_t)x + dx);
            if (skin_mask[nidx])
            {
              count++;
            }
          }
        }

        skin_mask_filt[idx] = (count >= 4U) ? 1U : 0U;
      }
    }
  }
}

static uint8_t DetectFaceHeuristicFromFrozenImage(const uint16_t *frame)
{
  uint32_t x, y;
  uint32_t best_area = 0U;

  BuildSkinMaskFromFrozenImage(frame);
  FilterSkinMask();

  for (y = 0U; y < FACE_H; y++)
  {
    for (x = 0U; x < FACE_W; x++)
    {
      uint32_t idx = y * FACE_W + x;

      if (skin_mask_filt[idx] == 1U)
      {
        uint32_t head = 0U;
        uint32_t tail = 0U;
        uint32_t area = 0U;
        uint32_t minx = x, maxx = x, miny = y, maxy = y;
        uint32_t bw, bh, bbox_area, fill_pct, cx, cy, aspect_pct;

        skin_mask_filt[idx] = 2U;
        bfs_queue[tail++] = (uint16_t)idx;

        while (head < tail)
        {
          uint32_t cur = bfs_queue[head++];
          uint32_t cx0 = cur % FACE_W;
          uint32_t cy0 = cur / FACE_W;
          int32_t dx, dy;

          area++;

          if (cx0 < minx) minx = cx0;
          if (cx0 > maxx) maxx = cx0;
          if (cy0 < miny) miny = cy0;
          if (cy0 > maxy) maxy = cy0;

          for (dy = -1; dy <= 1; dy++)
          {
            for (dx = -1; dx <= 1; dx++)
            {
              int32_t nx = (int32_t)cx0 + dx;
              int32_t ny = (int32_t)cy0 + dy;

              if ((nx >= 0) && (nx < (int32_t)FACE_W) &&
                  (ny >= 0) && (ny < (int32_t)FACE_H))
              {
                uint32_t nidx = (uint32_t)ny * FACE_W + (uint32_t)nx;

                if (skin_mask_filt[nidx] == 1U)
                {
                  skin_mask_filt[nidx] = 2U;
                  bfs_queue[tail++] = (uint16_t)nidx;
                }
              }
            }
          }
        }

        bw = maxx - minx + 1U;
        bh = maxy - miny + 1U;
        bbox_area = bw * bh;
        fill_pct = (bbox_area > 0U) ? ((100U * area) / bbox_area) : 0U;
        aspect_pct = (bh > 0U) ? ((100U * bw) / bh) : 0U;
        cx = (minx + maxx) / 2U;
        cy = (miny + maxy) / 2U;

        /*
         * Heuristiques visage probable :
         * - taille suffisante
         * - forme à peu près verticale/ovale
         * - pas trop bas dans l'image
         * - plutôt vers le centre
         * - remplissage raisonnable
         */
        if ((area >= 80U) &&
            (area <= 2500U) &&
            (bw >= 8U) &&
            (bh >= 10U) &&
            (aspect_pct >= 55U) &&
            (aspect_pct <= 140U) &&
            (fill_pct >= 35U) &&
            (fill_pct <= 95U) &&
            (cy < (FACE_H * 3U) / 4U) &&
            (cx > FACE_W / 6U) &&
            (cx < (FACE_W * 5U) / 6U))
        {
          if (area > best_area)
          {
            best_area = area;
          }
        }
      }
    }
  }

  return (best_area > 0U) ? 1U : 0U;
}

static void UpdateOverlayFromFrozenImage(void)
{
  char text[80] = {0};

  ApplySobelToCameraRGB565((const uint16_t *)CAMERA_FRAME_BUFFER,
                           (uint32_t *)LAYER1_ADDRESS,
                           LCD_WIDTH,
                           LCD_HEIGHT,
                           sobel_threshold,
                           sobel_color);

  BSP_LCD_SelectLayer(1);
  BSP_LCD_SetFont(&Font12);
  BSP_LCD_SetTextColor(LCD_COLOR_BLUE);
  BSP_LCD_SetBackColor(0x00000000U);

  sprintf(text, "Threshold:%lu  Face:%s   ",
          (unsigned long)sobel_threshold,
          face_detected ? "YES" : "NO");

  BSP_LCD_DisplayStringAtLine(1, (uint8_t *)text);
}

void BSP_CAMERA_FrameEventCallback(void)
{
  camera_frame_ready = 1U;
}

/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{
  /* USER CODE BEGIN 1 */
  ADC_ChannelConfTypeDef sConfig = {0};

  sConfig.Channel = ADC_CHANNEL_6;
  sConfig.Rank = ADC_REGULAR_RANK_1;
  sConfig.SamplingTime = ADC_SAMPLETIME_3CYCLES;
  /* USER CODE END 1 */

  /* MCU Configuration--------------------------------------------------------*/
  HAL_Init();
  SystemClock_Config();

  /* Initialize all configured peripherals */
  MX_GPIO_Init();
  MX_ADC3_Init();
  MX_DMA2D_Init();
  MX_FMC_Init();
  MX_I2C1_Init();
  MX_I2C3_Init();
  MX_LTDC_Init();
  MX_RTC_Init();
  MX_SPI2_Init();
  MX_TIM1_Init();
  MX_TIM2_Init();
  MX_TIM3_Init();
  MX_TIM5_Init();
  MX_TIM8_Init();
  MX_USART1_UART_Init();
  MX_USART6_UART_Init();
  MX_ADC1_Init();
  MX_DAC_Init();
  MX_UART7_Init();

  /* USER CODE BEGIN 2 */
  BSP_LCD_Init();

  BSP_LCD_LayerDefaultInit(0, LAYER0_ADDRESS);
  BSP_LCD_LayerDefaultInit(1, LAYER1_ADDRESS);
  BSP_LCD_DisplayOn();

  BSP_LCD_SelectLayer(0);
  BSP_LCD_Clear(LCD_COLOR_BLACK);

  BSP_LCD_SelectLayer(1);
  BSP_LCD_Clear(0x00000000U);
  BSP_LCD_SetTransparency(1, 255);

  HAL_ADC_ConfigChannel(&hadc3, &sConfig);

  if (BSP_CAMERA_Init(CAMERA_RESOLUTION) != CAMERA_OK)
  {
    Error_Handler();
  }

  /* Une seule photo au lancement */
  camera_frame_ready = 0U;
  BSP_CAMERA_SnapshotStart((uint8_t *)CAMERA_FRAME_BUFFER);

  while (camera_frame_ready == 0U)
  {
    HAL_Delay(1);
  }

  /* Affichage de l'image figée */
  CameraFrameToLayer0();

  /* Détection heuristique visage */
  face_detected = DetectFaceHeuristicFromFrozenImage((const uint16_t *)CAMERA_FRAME_BUFFER);
  sobel_color = face_detected ? 0xFF00FF00U : 0xFFFF0000U; /* vert si visage, rouge sinon */

  /* Premier affichage des contours */
  ReadSobelThresholdFromPot();
  UpdateOverlayFromFrozenImage();
  last_sobel_threshold = sobel_threshold;
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    ReadSobelThresholdFromPot();

    if ((sobel_threshold > last_sobel_threshold + 2U) ||
        (sobel_threshold + 2U < last_sobel_threshold))
    {
      UpdateOverlayFromFrozenImage();
      last_sobel_threshold = sobel_threshold;
    }

    HAL_Delay(30);
  }
  /* USER CODE END WHILE */

  /* USER CODE BEGIN 3 */
  /* USER CODE END 3 */
}

/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  HAL_PWR_EnableBkUpAccess();
  __HAL_RCC_PWR_CLK_ENABLE();
  __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_LSI | RCC_OSCILLATORTYPE_HSE;
  RCC_OscInitStruct.HSEState = RCC_HSE_ON;
  RCC_OscInitStruct.LSIState = RCC_LSI_ON;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSE;
  RCC_OscInitStruct.PLL.PLLM = 25;
  RCC_OscInitStruct.PLL.PLLN = 400;
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV2;
  RCC_OscInitStruct.PLL.PLLQ = 9;

  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  if (HAL_PWREx_EnableOverDrive() != HAL_OK)
  {
    Error_Handler();
  }

  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                              | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV4;
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV2;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_6) != HAL_OK)
  {
    Error_Handler();
  }
}

/**
  * @brief  Period elapsed callback in non blocking mode
  * @note   This function is called when TIM6 interrupt took place, inside
  * HAL_TIM_IRQHandler(). It makes a direct call to HAL_IncTick() to increment
  * a global variable "uwTick" used as application time base.
  * @param  htim : TIM handle
  * @retval None
  */
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  if (htim->Instance == TIM6)
  {
    HAL_IncTick();
  }
}

/**
  * @brief  This function is executed in case of error occurrence.
  * @retval None
  */

{
  __disable_irq();
  while (1)
  {
  }
}
